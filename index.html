2. Before → After

Before
  • 배포·CMS·BCS 절차가 엑셀 문서에 정리되어 있으나 부족·분산
  • 산출물은 고정 틀이 있으나, 공지 등 실무 절차가 한곳에 모여 있지 않음
  • 담당 변경·온보딩 시 Git→CMS→BCS 전체 흐름을 한눈에 보기 어려움
  • 파일(엑셀) 중심이라 검색·공유·개정 이력 관리가 불편
After
  • Git · CMS · BCS를 통합 운영가이드로 한곳에 정리·보완
  • 기존 산출물 틀·고정값을 가이드에 반영하고, 공지 템플릿·MR 규칙까지 연결
  • 인수인계·온보딩용 end-to-end 절차를 웹에서 한 경로로 확인
  • 웹 가이드로 검색·공유·개정이 가능하고, UI 컴포넌트 표준까지 함께 제공

// 프론트엔드 Validation — aria-invalid + .lg-form__error-msg 바인딩
function setFieldError(field, message) {
  const group = field.closest('.lg-form__group');
  const errorEl = group.querySelector('.lg-form__error-msg');
  field.setAttribute('aria-invalid', 'true');
  errorEl.textContent = message;
  errorEl.hidden = false;
}

function clearFieldError(field) {
  const group = field.closest('.lg-form__group');
  const errorEl = group.querySelector('.lg-form__error-msg');
  field.removeAttribute('aria-invalid');
  errorEl.textContent = '';
  errorEl.hidden = true;
}

form.addEventListener('submit', (e) => {
  e.preventDefault();
  if (!userName.value.trim()) {
    setFieldError(userName, '이름을 입력해 주세요.');
    return;
  }
  clearFieldError(userName);
});

JSTL 가이드

<nav class="lg-pagination" data-lg-component="pagination" aria-label="페이지 네비게이션">
  <ul class="lg-pagination__list">
    <li class="lg-pagination__item">
      <c:choose>
        <c:when test="${prevDisabledYn eq 'Y'}">
          <button type="button" class="lg-pagination__btn lg-pagination__btn--prev" disabled>이전</button>
        </c:when>
        <c:otherwise>
          <a href="${baseUrl}${currentPage - 1}" class="lg-pagination__btn lg-pagination__btn--prev"
             data-lg-toggle="page" data-lg-target="${currentPage - 1}">이전</a>
        </c:otherwise>
      </c:choose>
    </li>
    <c:forEach var="page" items="${pageList}">
      <li class="lg-pagination__item">
        <c:choose>
          <c:when test="${page.type eq 'ellipsis'}">
            <span class="lg-pagination__ellipsis">&hellip;</span>
          </c:when>
          <c:when test="${page.activeYn eq 'Y'}">
            <button type="button" class="lg-pagination__btn lg-pagination__btn--active"
                    data-lg-toggle="page" data-lg-target="${page.pageNum}" aria-current="page">${page.pageNum}</button>
          </c:when>
          <c:otherwise>
            <a href="${baseUrl}${page.pageNum}" class="lg-pagination__btn"
               data-lg-toggle="page" data-lg-target="${page.pageNum}">${page.pageNum}</a>
          </c:otherwise>
        </c:choose>
      </li>
    </c:forEach>
    <li class="lg-pagination__item">
      <c:choose>
        <c:when test="${nextDisabledYn eq 'Y'}">
          <button type="button" class="lg-pagination__btn lg-pagination__btn--next" disabled>다음</button>
        </c:when>
        <c:otherwise>
          <a href="${baseUrl}${currentPage + 1}" class="lg-pagination__btn lg-pagination__btn--next"
             data-lg-toggle="page" data-lg-target="${currentPage + 1}">다음</a>
        </c:otherwise>
      </c:choose>
    </li>
  </ul>
</nav>
