Alert Case — 타이틀 + 설명

미리보기 (Preview)

실서버에서는 lgkorUi.alert + jQuery 스크립트로 동작합니다.

접근성 및 스크립트 필수 규격

  • data-id="#laypop2" — 모달 레이어 팝업 고유 ID 연동
  • data-control="modal" — 표준 모달 제어 이벤트 바인딩
  • lgkorUi.alert — 사내 UI 라이브러리 커스텀 얼럿 호출
  • typeClass: 'type2' — 오류/안내 전용 시각 스타일 지정

HTML 코드

<!-- DOM -->
<button type="button" data-id="#laypop2" data-control="modal" class="btn"><span>Alert- 타이틀 + 설명</span></button>

스크립트(Script) 가이드

<script>
$(function() {
  $('.btn').on('click', function(e) {
    var id = $(e.currentTarget).data('id');
    var obj = { title: '', typeClass: '', ok: function() {} };
    var desc = '';

    if (id == "#laypop2") {
      obj = $.extend(obj, { typeClass: 'type2', title: '인증이 실패했습니다.' });
      desc = '<p>자세한 사항은 1577-4090 으로<br>문의부탁드립니다.</p>';
    }
    lgkorUi.alert(desc, obj);
  });
});
</script>

JSTL 및 유효성 검사(Validation) 가이드

<%@ taglib prefix="c" uri="jakarta.tags.core" %>

<form class="lg-form" data-lg-component="form" method="post" action="${submitUrl}" novalidate>
  <div class="lg-form__group">
    <label for="userName" class="lg-form__label">
      이름 <span class="lg-form__required" aria-hidden="true">*</span>
    </label>
    <input type="text" id="userName" name="userName" class="lg-form__input"
           value="<c:out value='${formData.userName}'/>"
           required aria-required="true"
           <c:if test="${not empty fieldErrors.userName}">aria-invalid="true"</c:if>>
    <c:if test="${not empty fieldErrors.userName}">
      <p class="lg-form__error-msg" role="alert" aria-live="assertive">
        <c:out value="${fieldErrors.userName}"/>
      </p>
    </c:if>
  </div>
</form>
// 프론트엔드 Validation — aria-invalid + .lg-form__error-msg 바인딩
function setFieldError(field, message) {
  const group = field.closest('.lg-form__group');
  const errorEl = group.querySelector('.lg-form__error-msg');
  field.setAttribute('aria-invalid', 'true');
  errorEl.textContent = message;
  errorEl.hidden = false;
}

function clearFieldError(field) {
  const group = field.closest('.lg-form__group');
  const errorEl = group.querySelector('.lg-form__error-msg');
  field.removeAttribute('aria-invalid');
  errorEl.textContent = '';
  errorEl.hidden = true;
}

form.addEventListener('submit', (e) => {
  e.preventDefault();
  if (!userName.value.trim()) {
    setFieldError(userName, '이름을 입력해 주세요.');
    return;
  }
  clearFieldError(userName);
});
